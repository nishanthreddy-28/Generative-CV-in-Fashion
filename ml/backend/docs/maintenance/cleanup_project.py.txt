actions=[]
def safe(p):
    p=Path(p).absolute()
    if not p.resolve().is_relative_to(ROOT) or p.resolve()==ROOT: raise RuntimeError(f'Unsafe target: {p}')
    return p
def move(src,dst):
    src,dst=safe(src),safe(dst)
    if not src.exists(): return
    if dst.exists(): raise RuntimeError(f'Destination exists: {dst}')
    dst.parent.mkdir(parents=True,exist_ok=True)
    shutil.move(str(src),str(dst)); actions.append({'move':str(src.relative_to(ROOT)),'to':str(dst.relative_to(ROOT))})
def remove(p,reason):
    p=safe(p)
    if not p.exists(): return
    if p.is_dir(): shutil.rmtree(p)
    else: p.unlink()
    actions.append({'removed':str(p.relative_to(ROOT)),'reason':reason})
def read(p):return (B/p).read_text(encoding='utf-8-sig')
def write(p,s): (B/p).write_text(s,encoding='utf-8')
def replace(p,a,b):
    s=read(p)
    if a not in s: raise RuntimeError(f'Missing edit in {p}: {a[:60]}')
    write(p,s.replace(a,b))
def drop_methods(p,names):
    s=read(p); lines=s.splitlines(keepends=True); nodes=ast.walk(ast.parse(s))
    spans=[(n.lineno-1,n.end_lineno) for n in nodes if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) and n.name in names]
    for start,end in sorted(spans,reverse=True): del lines[start:end]
    write(p,''.join(lines))
