def imports(s):
    lines=s.splitlines(keepends=True); edits=[]
    for node in ast.walk(ast.parse(s)):
        if isinstance(node,ast.ImportFrom) and node.module and not node.level:
            replacements=[]; changed=False
            for a in node.names:
                whole=node.module+'.'+a.name
                if whole in mapping:
                    module,name=mapping[whole].rsplit('.',1)
                    alias=a.asname or (a.name if a.name!=name else None)
                    replacements.append(f'from {module} import {name}'+(f' as {alias}' if alias else ''))
                    changed=True
                else:
                    module=mapping.get(node.module,node.module)
                    changed |= module!=node.module
                    replacements.append(f'from {module} import {a.name}'+(f' as {a.asname}' if a.asname else ''))
            if changed:
                indent=' '*node.col_offset
                edits.append((node.lineno-1,node.end_lineno,''.join(indent+r+'\n' for r in replacements)))
    for a,b,r in sorted(edits,reverse=True):lines[a:b]=[r]
    return ''.join(lines)
